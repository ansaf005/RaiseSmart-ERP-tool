function sChoose(type){
var key=S.chooseKey;var isFresh=/F$/.test(key);var n=+key.match(/\d+/)[0];
var h='<span class="crumb" onclick="S.chooseKey=null;if(window.syncState) window.syncState(); go(S.tab)">&larr; Back to '+TYPES[type]+'s</span>';
if(!isFresh&&actStatus(n)==='Not Yet Started'){
var qs=S.questions.filter(function(q){return q.type===type&&q.num===n&&q.campus===S.user.campus&&q.published;});
h+='<h2>'+TYPES[type]+' '+n+' - Problem Statements (Preview)</h2><div class="sub">Choosing has not opened yet.</div><div class="grid2">';
qs.forEach(function(q){h+='<div class="teamcard"><b>'+esc(q.company)+'</b><div style="margin:6px 0">'+esc(q.statement)+'</div><div class="sub">'+esc(q.details)+'</div><div class="slotrow">'+q.slots.map(function(sp){return '<span class="slot">'+SPECSHORT[sp]+'</span>';}).join('')+'</div></div>';});
return h+'</div>';
}
var probs=S.problems[key];var mine=curTeam(type);var my=userSpecs();
h+='<h2>'+TYPES[type]+' '+n+' - Choose Your Challenge</h2>'
+'<div class="sub">Each challenge takes exactly <b>5 members</b> and each seat is reserved for a specialization fixed by the Coordinator. Once you choose, all other challenges grey out. When 5 members choose the same challenge, they automatically become one team.</div>'
+'<div class="noteinfo">Showing only the challenges that match your chosen specialization(s): <b>'+my.map(esc).join(', ')+'</b>. Challenges without a seat for your specialization(s) are not listed in your page.</div>';
if(mine){h+='<div class="noteinfo">You have chosen <b>'+esc(mine.team.company)+'</b> ('+mine.team.team+'). All other challenges are locked for you.</div>';}
h+='<div class="grid2">';
var shown=0;
probs.forEach(function(p,ti){
var isMine=mine&&mine.ti===ti;
var matches=p.slots.some(function(sl){return my.indexOf(sl.spec)>=0;});
if(!matches&&!isMine)return; /* challenge not listed for this student's specializations */
shown++;
var count=p.members.filter(function(m){return !m.cr;}).length;
var full=count>=5;
var cls=isMine?'teamcard mine':((full||mine)?'teamcard grey':'teamcard');
var members=p.members.map(function(m){var s=stu(m.sid);return '<div class="member">'+esc(s.name)+' <span class="pill">'+s.sid+'</span> <span class="pill">'+SPECSHORT[s.spec]+'</span>'+tlStar(m.sid,p.tl)+(m.cr?' <span class="chip chip-r">Critical Resource</span>':'')+'</div>';}).join('');
var slots=p.slots.map(function(sl){return '<span class="slot'+(sl.sid?' filled':'')+'">'+SPECSHORT[sl.spec]+(sl.sid?(' - '+esc(stu(sl.sid).name.split(' ')[0])):' - Open')+'</span>';}).join('');
var action='';
if(isMine)action='<span class="chip chip-b">YOUR TEAM</span>';
else if(full)action='<span class="chip chip-gr">TEAM FULL - LOCKED</span>';
else if(mine)action='<span class="chip chip-gr">Locked (you already chose)</span>';
else action='<button class="btn" onclick="chooseProblem(\''+key+'\','+ti+')">Choose This Challenge</button>';
h+='<div class="'+cls+'"><div style="display:flex;justify-content:space-between;align-items:center"><b>'+esc(p.company)+'</b><span class="chip '+(full?'chip-g':'chip-a')+'">'+count+'/5 members</span></div>'
+'<div class="pill" style="margin:4px 0;display:inline-block">'+p.team+'</div>'
+'<div style="margin:6px 0;font-size:13px">'+esc(p.statement)+'</div>'
+'<div class="sub" style="margin-bottom:6px">'+esc(p.details)+'</div>'
+'<div style="font-size:11px;font-weight:700;color:var(--mut)">SPECIALIZATION SEATS (fixed by the Coordinator)</div><div class="slotrow">'+slots+'</div>'
+'<div style="font-size:11px;font-weight:700;color:var(--mut);margin-top:6px">WHO HAS CHOSEN</div>'+(members||'<div class="sub">No one yet.</div>')
+'<div style="margin-top:10px">'+action+'</div></div>';
});
h+='</div>';
if(!shown)h+='<div class="notewarn">No challenges match your chosen specialization(s) right now. New challenges that include your specialization will appear here automatically.</div>';
return h;
}
function chooseProblem(key,ti){
var type=key.slice(0,2);
if(curTeam(type)){toast('You have already chosen a challenge for this activity.');return;}
var p=S.problems[key][ti];
var my=userSpecs();
var slot=null;
for(var i=0;i<p.slots.length;i++){if(!p.slots[i].sid&&my.indexOf(p.slots[i].spec)>=0){slot=p.slots[i];break;}}
if(!slot){
var open=p.slots.filter(function(sl){return !sl.sid;}).map(function(sl){return sl.spec;});
toast(open.length?('Open seat requires: '+open.join(', ')+'. Your specialization(s): '+my.join(', ')+'.'):'This team is already full (5/5).');
return;
}
slot.sid=S.user.sid;
p.members.push({sid:S.user.sid,cr:false});
var count=p.members.filter(function(m){return !m.cr;}).length;
if(count>=5){toast('You joined '+p.company+'. Team is complete (5/5) - '+p.team+' has been formed automatically! Now assign a Team Leader in My Team.');}
else{toast('You joined '+p.company+' ('+count+'/5).');}
if(window.syncState) window.syncState(); go(S.tab);
}

function assignTL(key,ti,sid){
var p=S.problems[key][ti];
if(p.tl){toast('Team Leader already assigned.');return;}
if(!confirm('Assign '+stu(sid).name+' as Team Leader? (All 5 members confirm this choice together)'))return;
p.tl=sid;
toast('All 5 members confirmed: '+stu(sid).name+' is now the Team Leader ★ of '+p.team+'.');
if(window.syncState) window.syncState(); go(S.tab);
}

function tlGate(type){
var mine=curTeam(type);
if(!mine)return {html:'<div class="notewarn">You are not in a '+TYPES[type]+' team yet. Choose a challenge first.</div>'};

return {mine:mine,ok:true};
}

function buildStudentCharts(){
if(!window.Chart)return;
var acts={1:[1,2],2:[3,4],3:[5,6]}[S.sdSem]||[1,2];
function mm(t,n,r){
var mine=teamOfUser(t,n,S.user.sid);if(!mine)return null;
var k=actKey(t,n,S.user.campus);var rec=(S.marks[k]&&S.marks[k][r])?S.marks[k][r][mine.ti]:null;
if(!rec||rec.status!=='published')return null;
var v=rec.entries[S.user.sid];return v==null?null:v;
}
function mk(id,cfg){var el=document.getElementById(id);if(el)S.charts.push(new Chart(el,cfg));}
var cols=['#1a56db','#0e9f6e'];
mk('cs1',{type:'bar',data:{labels:REVIEWS.map(function(r){return r[1];}),datasets:acts.map(function(n,i){
return {label:TYPES[S.sdType]+' '+n,data:REVIEWS.map(function(r){return mm(S.sdType,n,r[0])||0;}),backgroundColor:cols[i]};})},
options:{scales:{y:{beginAtZero:true,max:100}}}});
function avgFor(t,n){var tot=0,c=0;REVIEWS.forEach(function(r){var v=mm(t,n,r[0]);if(v!=null){tot+=v;c++;}});return c?Math.round(tot/c):0;}
mk('cs2',{type:'bar',data:{labels:acts.map(function(n){return 'Activity '+n;}),datasets:[
{label:'Business Simulation',data:acts.map(function(n){return avgFor('BS',n);}),backgroundColor:'#1a56db'},
{label:'Live Project',data:acts.map(function(n){return avgFor('LP',n);}),backgroundColor:'#0e9f6e'}]},
options:{scales:{y:{beginAtZero:true,max:100}}}});
}
